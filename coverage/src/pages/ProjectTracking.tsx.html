type SimpleSubtaskInputProps = {
  onSubmit: (value: string) => void;
};
 
const SimpleSubtaskInput: React.FC<SimpleSubtaskInputProps> = React.memo(({ onSubmit }) => {
  const inputRef = useRef<HTMLInputElement>(null);
  const latestValueRef = useRef('');
 
  const handleKeyDown = useCallback((e: React.KeyboardEvent<HTMLInputElement>) => {
    e.stopPropagation();
    const inputEl = inputRef.current;
    if (!inputEl) return;
 
    if (e.key === 'Enter') {
      const trimmed = inputEl.value.trim();
      if (!trimmed) {
        return;
      }
 
      e.preventDefault();
      e.stopPropagation();
 
      onSubmit(trimmed);
      inputEl.value = '';
      latestValueRef.current = '';
 
      requestAnimationFrame(() => {
        inputEl.focus({ preventScroll: true });
      });
    }
 
    if (e.key === 'Escape') {
      e.preventDefault();
      inputEl.value = '';
      latestValueRef.current = '';
      inputEl.blur();
    }
  }, [onSubmit]);
 
  const handleChange = useCallback((e: React.ChangeEvent<HTMLInputElement>) => {
    e.stopPropagation();
    latestValueRef.current = inputRef.current?.value ?? '';
  }, []);
 
  const handleFocus = useCallback((e: React.FocusEvent<HTMLInputElement>) => {
    e.stopPropagation();
    if (inputRef.current) {
      inputRef.current.scrollIntoView = () => {};
    }
  }, []);
 
  const handleBlur = useCallback((e: React.FocusEvent<HTMLInputElement>) => {
    e.stopPropagation();
    const el = inputRef.current;
    requestAnimationFrame(() => {
      if (document.activeElement !== el && el) {
        const cursor = el.value.length;
        el.focus({ preventScroll: true });
        el.setSelectionRange?.(cursor, cursor);
      }
    });
  }, []);
 
  useEffect(() => {
    const el = inputRef.current;
    if (!el) return;
 
    if (document.activeElement !== el) {
      const cursor = el.value.length;
      el.focus({ preventScroll: true });
      el.setSelectionRange?.(cursor, cursor);
    }
  });
 
  return (
    <div className="mt-2">
      <input
        ref={inputRef}
        type="text"
        placeholder="Add subtask..."
        onChange={handleChange}
        onKeyDown={handleKeyDown}
        onFocus={handleFocus}
        onBlur={handleBlur}
        className="block w-full rounded-md border border-gray-300 bg-white px-3 py-2 text-sm focus:border-blue-500 focus:outline-none focus:ring-2 focus:ring-blue-500 dark:border-slate-600 dark:bg-slate-800 dark:text-white dark:focus:border-blue-400 dark:focus:ring-blue-400"
        autoComplete="off"
        spellCheck="false"
      />
    </div>
  );
});
